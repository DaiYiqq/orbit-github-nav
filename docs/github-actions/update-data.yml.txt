name: 自动收录 GitHub 仓库

# GitHub 有数亿仓库，客户端无法全量下载。
# 本工作流每周按「星标分层 + 语言 + 主题」自动收录高价值仓库头部，
# 生成离线索引；索引外的长尾仓库由前端「GitHub 全站」实时搜索覆盖。

on:
  schedule:
    - cron: '0 3 * * 1'          # 每周一 03:00 UTC
  workflow_dispatch:

permissions:
  contents: write
  actions: write

concurrency:
  group: update-data
  cancel-in-progress: false

jobs:
  index:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '22'

      - name: 抓取并自动分类
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: node scripts/fetch-repos.mjs

      - name: 校验索引
        run: |
          node -e "
            const m = require('./site/data/meta.json');
            if (!m.count || m.count < 1000) throw new Error('索引数量异常: ' + m.count);
            console.log('索引校验通过:', m.count, '个仓库,', m.categories.length, '个分类');
          "

      - name: 提交更新后的索引
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          git config user.name "orbit-index-bot"
          git config user.email "orbit-index-bot@users.noreply.github.com"
          git add site/data/
          if git diff --staged --quiet; then
            echo "数据无变化，跳过提交"
          else
            git commit -m "chore: 自动更新 GitHub 仓库索引"
            git push
            # GITHUB_TOKEN 触发的 push 不会再触发其它工作流，显式触发 Pages 部署
            gh workflow run deploy-pages.yml || echo "请检查 Pages 工作流权限"
          fi
